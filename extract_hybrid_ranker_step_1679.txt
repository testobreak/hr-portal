"    def graph_score(self, session, symbol: Symbol | None, context: Dict[str, Any] = None) -> float:\n        \"\"\"\n        Gives a 50 point boost if candidate has code relationships to active context symbols.\n        Otherwise falls back to general codebase connectivity centrality.\n        \"\"\"\n        if not symbol:\n            return 0.0\n\n        # Direct context connection boost\n        if context:\n            active_symbol_name = context.get(\"active_symbol\")\n            if active_symbol_name:\n                active_symbol = session.query(Symbol).filter(Symbol.qualified_name == active_symbol_name).first()\n                if active_symbol:\n                    exists = session.query(SymbolRelationship).filter(\n                        SymbolRelationship.repository_id == symbol.repository_id,\n                        (\n                            (SymbolRelationship.source_symbol_id == active_symbol.id) &\n                            (SymbolRelationship.target_symbol_id == symbol.id)\n                        ) |\n                        (\n                            (SymbolRelationship.source_symbol_id == symbol.id) &\n                            (SymbolRelationship.target_symbol_id == active_symbol.id)\n                        )\n                    ).first()\n                    if exists:\n                        return 50.0\n\n        # General codebase connectivity fallback\n        inbound = session.query(SymbolRelationship).filter(SymbolRelationship.target_symbol_id == symbol.id).count()\n        outbound = session.query(SymbolRelationship).filter(SymbolRelationship.source_symbol_id == symbol.id).count()\n        total_connections = inbound + outbound\n        return min(50.0, total_connections * 10.0)"
